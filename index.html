Firebase 권한 설정과 로그인 상태를 확인해주세요.\n' + (err && err.message ? err.message : '')); } } async function insertEntry(entry) { const user = await ensureAuthed(); const col = collection(); if (!col) return; await col.doc(entry.id).set({ sentences: entry.sentences, createdAt: firebase.firestore.Timestamp.fromDate(new Date(entry.createdAt)), ownerUid: user.uid }); } async function appendEntrySentences(entryId, newPairs) { if (!Array.isArray(newPairs) || newPairs.length === 0) { throw new Error('newPairs must be a non-empty array'); } const user = await ensureAuthed(); const col = collection(); if (!col) throw new Error('Room not selected'); const entryRef = col.doc(entryId); return firebase.firestore().runTransaction(async tx => { const doc = await tx.get(entryRef); if (!doc.exists) throw new Error('Entry not found'); const data = doc.data() || {}; const entry = { ownerUid: data.ownerUid || user.uid }; if (!isEntryOwner(entry, user)) { throw new Error('권한 없음: 이 항목에 추가할 수 없습니다.'); } const current = Array.isArray(data.sentences) ? data.sentences : []; const updated = current.concat(newPairs.map(p => ({ ko: p.ko, en: p.en }))); tx.update(entryRef, { sentences: updated, updatedAt: firebase.firestore.Timestamp.now() }); return { before: current.length, after: updated.length }; }); }
